let chart;
let currentSelection = null;

function initChart() {
    const ctx = document.getElementById('taxPieChart').getContext('2d');
    
    chart = new Chart(ctx, {
        type: 'pie',
        data: {
            labels: taxData.map(item => item.name),
            datasets: [{
                data: taxData.map(item => item.percentage),
                backgroundColor: taxData.map(item => item.color),
                borderWidth: 3,
                borderColor: '#ffffff',
                hoverOffset: 15,
                offset: [0, 0, 0, 0, 0, 0]
            }]
        },
        options: {
            responsive: true,
            maintainAspectRatio: false,
            plugins: {
                legend: {
                    display: false
                },
                tooltip: {
                    callbacks: {
                        label: function(context) {
                            return context.label + ': ' + context.parsed + '%';
                        }
                    }
                }
            },
            onClick: (event, elements) => {
                if (elements.length > 0) {
                    const index = elements[0].index;
                    showDescription(index);
                }
            }
        }
    });

    createLegend();
}

function createLegend() {
    const legendContainer = document.getElementById('legend');
    legendContainer.innerHTML = '';

    taxData.forEach((item, index) => {
        const legendItem = document.createElement('div');
        legendItem.className = 'legend-item';
        legendItem.onclick = () => showDescription(index);

        legendItem.innerHTML = `
            <div class="legend-color" style="background-color: ${item.color}"></div>
            <span class="legend-label">${item.name}</span>
            <span class="legend-percentage">${item.percentage}%</span>
        `;

        legendContainer.appendChild(legendItem);
    });
}

function showDescription(index) {
    const item = taxData[index];
    const descriptionContainer = document.getElementById('descriptionContainer');
    
    descriptionContainer.innerHTML = `
        <div class="description-title">${item.name}</div>
        <div class="description-percentage" style="color: ${item.color}">${item.percentage}%</div>
        <div style="font-size: 1.2rem; font-weight: 600; color: #4a5568; margin-bottom: 1rem;">${item.amount}</div>
        <div class="description-text">${item.description}</div>
    `;

    // Highlight the selected slice
    if (chart) {
        // Reset all offsets to 0
        const offsets = new Array(taxData.length).fill(0);
        // Set offset for selected slice
        offsets[index] = 15;
        
        chart.data.datasets[0].offset = offsets;
        chart.setActiveElements([{
            datasetIndex: 0,
            index: index,
        }]);
        chart.update();
    }

    currentSelection = index;
}

function togglePopup(element, event) {
    event.stopPropagation();
    const popup = element.querySelector('.popup');
    const isCurrentlyVisible = popup.style.display === 'block';
    const isPinned = element.getAttribute('data-pinned') === 'true';
    
    // Close all other popups first
    document.querySelectorAll('.popup').forEach(p => {
        if (p !== popup) {
            p.style.display = 'none';
            if (p.parentElement) {
                p.parentElement.setAttribute('data-pinned', 'false');
            }
        }
    });
    
    // Toggle this popup
    if (!isCurrentlyVisible || !isPinned) {
        popup.style.display = 'block';
        element.setAttribute('data-pinned', 'true');
    } else {
        popup.style.display = 'none';
        element.setAttribute('data-pinned', 'false');
    }
}

function showPopup(element) {
    const popup = element.querySelector('.popup');
    const isPinned = element.getAttribute('data-pinned') === 'true';
    
    if (!isPinned) {
        popup.style.display = 'block';
    }
}

function hidePopup(element) {
    const popup = element.querySelector('.popup');
    const isPinned = element.getAttribute('data-pinned') === 'true';
    
    if (!isPinned) {
        popup.style.display = 'none';
    }
}

// Prevent popup from closing when hovering over it
document.addEventListener('DOMContentLoaded', function() {
    document.querySelectorAll('.popup').forEach(popup => {
        popup.addEventListener('mouseenter', function(e) {
            e.stopPropagation();
            const parent = this.parentElement;
            if (parent && parent.getAttribute('data-pinned') === 'true') {
                // Keep popup open if pinned
                this.style.display = 'block';
            }
        });
        
        popup.addEventListener('mouseleave', function(e) {
            const parent = this.parentElement;
            if (parent && parent.getAttribute('data-pinned') !== 'true') {
                this.style.display = 'none';
            }
        });
    });
});

function initHeadingAnimations() {
    // Trigger heading animations after a short delay
    setTimeout(() => {
        const fiscalYear = document.querySelector('.fiscal-year-text');
        const mainHeading = document.querySelector('.main-heading');
        
        if (fiscalYear) {
            fiscalYear.style.opacity = '1';
            fiscalYear.style.transform = 'translateY(0)';
        }
        
        if (mainHeading) {
            mainHeading.style.opacity = '1';
            mainHeading.style.transform = 'translateY(0)';
        }
    }, 500);
}

function initScrollAnimations() {
    const observer = new IntersectionObserver(
        entries => entries.forEach(e => e.target.classList.toggle('visible', e.isIntersecting)),
        { threshold: 0.1 }
    );
    
    document.querySelectorAll('.content-wrapper, .where-taxes-go-section, .bar-charts-section')
        .forEach(el => observer.observe(el));
}

// Helper function to create stacked bar segments
function createSegment({ pct, total, color, label }) {
    const seg = document.createElement('div');
    seg.style.cssText = `
        height: ${pct/total*100}%;
        background: ${color};
        cursor: pointer;
        transition: all .2s ease;
        position: relative;
    `;

    const popup = document.createElement('div');
    popup.className = 'seg-popup';
    popup.textContent = `${label}: ${pct}%`;
    seg.appendChild(popup);

    seg.addEventListener('mouseenter', () => {
        popup.style.display = 'block';
        seg.style.filter = 'brightness(1.1)';
        seg.style.transform = 'scale(1.05)';
        seg.parentElement.style.zIndex = '100';
    });
    
    seg.addEventListener('mouseleave', () => {
        popup.style.display = 'none';
        seg.style.filter = '';
        seg.style.transform = '';
        seg.parentElement.style.zIndex = '2';
    });
    
    return seg;
}

function toggleExplanation() {
    const explanationDiv = document.getElementById('moreExplanation');
    const button = document.getElementById('moreExplanationBtn');
    
    explanationDiv.classList.toggle('explanation-visible');
    
    if (explanationDiv.classList.contains('explanation-visible')) {
        button.textContent = 'Close';
        button.style.background = '#4A6B5C';
    } else {
        button.textContent = 'Example';
        button.style.background = '#6B8E7F';
    }
}
